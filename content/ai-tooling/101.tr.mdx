## LLM Sağlayıcı API'leri

### OpenAI

```python
from openai import OpenAI

client = OpenAI(api_key="your-key")

response = client.chat.completions.create(
    model="gpt-4",
    messages=[
        {"role": "system", "content": "Yardımsever bir asistansınız."},
        {"role": "user", "content": "Kuantum hesaplamayı açıklayın."}
    ],
    temperature=0.7,
    max_tokens=500
)

print(response.choices[0].message.content)
```

### Anthropic (Claude)

```python
from anthropic import Anthropic

client = Anthropic(api_key="your-key")

response = client.messages.create(
    model="claude-3-opus-20240229",
    max_tokens=1024,
    messages=[
        {"role": "user", "content": "Kuantum hesaplamayı açıklayın."}
    ]
)

print(response.content[0].text)
```

### Yerel Modeller (Ollama)

```python
import ollama

response = ollama.chat(
    model='llama2',
    messages=[
        {'role': 'user', 'content': 'Kuantum hesaplamayı açıklayın.'}
    ]
)

print(response['message']['content'])
```

### Temel Zincir

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

# Bileşenler oluştur
llm = ChatOpenAI(model="gpt-4")
prompt = ChatPromptTemplate.from_messages([
    ("system", "Yardımsever bir asistansınız."),
    ("user", "{input}")
])
output_parser = StrOutputParser()

# Zincir oluştur
chain = prompt | llm | output_parser

# Çalıştır
result = chain.invoke({"input": "AI nedir?"})
print(result)
```

### RAG Zinciri

```python
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain_core.runnables import RunnablePassthrough

# Alıcı kurulumu
vectorstore = Chroma(embedding_function=OpenAIEmbeddings())
retriever = vectorstore.as_retriever()

# RAG promptu
rag_prompt = ChatPromptTemplate.from_template("""
Bağlama göre yanıt verin:
Bağlam: {context}
Soru: {question}
""")

# RAG zinciri oluştur
rag_chain = (
    {"context": retriever, "question": RunnablePassthrough()}
    | rag_prompt
    | llm
    | output_parser
)
```

### Chroma Örneği

```python
import chromadb
from chromadb.utils import embedding_functions

# Başlat
client = chromadb.Client()
embedding_fn = embedding_functions.OpenAIEmbeddingFunction(
    api_key="your-key"
)

# Koleksiyon oluştur
collection = client.create_collection(
    name="documents",
    embedding_function=embedding_fn
)

# Belgeler ekle
collection.add(
    documents=["AI teknolojiyi dönüştürüyor", "ML AI'ın bir alt kümesidir"],
    ids=["doc1", "doc2"]
)

# Sorgula
results = collection.query(
    query_texts=["Yapay zeka nedir?"],
    n_results=2
)
```
