## Case Study 1: Multi-Agent Collaboration System

### Challenge

Build a system where multiple specialized agents collaborate on complex tasks.

<Mermaid chart={`
flowchart TB
    Task[Complex Task] --> Orchestrator[Orchestrator Agent]
    
    Orchestrator --> Research[Research Agent]
    Orchestrator --> Code[Code Agent]
    Orchestrator --> Review[Review Agent]
    
    Research --> Share[(Shared Memory)]
    Code --> Share
    Review --> Share
    
    Share --> Orchestrator
    Orchestrator --> Final[Final Output]
`} />

### Implementation

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict, Annotated
import operator

class CollaborativeState(TypedDict):
    task: str
    research: Annotated[list, operator.add]
    code: Annotated[list, operator.add]
    reviews: Annotated[list, operator.add]
    final_output: str
    iteration: int

class MultiAgentSystem:
    def __init__(self):
        self.research_agent = self._create_agent("research")
        self.code_agent = self._create_agent("coding")
        self.review_agent = self._create_agent("review")
        self.orchestrator = self._create_orchestrator()
        
        self.graph = self._build_graph()
    
    def _create_agent(self, specialty: str):
        prompts = {
            "research": """You are a research specialist.
                Given the task and any previous context, gather information.
                Task: {task}
                Previous research: {research}
                Provide new findings:""",
            "coding": """You are a coding specialist.
                Based on research, write code.
                Task: {task}
                Research: {research}
                Write implementation:""",
            "review": """You are a code reviewer.
                Review this code for issues.
                Code: {code}
                Provide detailed review:"""
        }
        return ChatOpenAI(model="gpt-4") | StrOutputParser()
    
    def _build_graph(self):
        workflow = StateGraph(CollaborativeState)
        
        workflow.add_node("orchestrate", self.orchestrate)
        workflow.add_node("research", self.do_research)
        workflow.add_node("code", self.do_coding)
        workflow.add_node("review", self.do_review)
        workflow.add_node("synthesize", self.synthesize)
        
        workflow.set_entry_point("orchestrate")
        
        workflow.add_conditional_edges(
            "orchestrate",
            self.route_next,
            {
                "research": "research",
                "code": "code",
                "review": "review",
                "done": "synthesize"
            }
        )
        
        workflow.add_edge("research", "orchestrate")
        workflow.add_edge("code", "orchestrate")
        workflow.add_edge("review", "orchestrate")
        workflow.add_edge("synthesize", END)
        
        return workflow.compile()
    
    def run(self, task: str):
        initial_state = {
            "task": task,
            "research": [],
            "code": [],
            "reviews": [],
            "final_output": "",
            "iteration": 0
        }
        return self.graph.invoke(initial_state)
```

## Case Study 2: Self-Healing Pipeline

### Challenge

Create a pipeline that automatically detects and recovers from failures.

<Mermaid chart={`
flowchart TB
    Input[Input] --> Process[Process Step]
    Process --> Check{Success?}
    Check -->|Yes| Next[Next Step]
    Check -->|No| Diagnose[Diagnose Error]
    Diagnose --> Fix[Auto-Fix]
    Fix --> Retry[Retry]
    Retry --> Process
    Retry --> Escalate[Escalate to Human]
`} />

```python
from tenacity import retry, stop_after_attempt, wait_exponential
from langchain_core.runnables import RunnableLambda

class SelfHealingPipeline:
    def __init__(self):
        self.llm = ChatOpenAI(model="gpt-4")
        self.error_handler = self._create_error_handler()
    
    def _create_error_handler(self):
        return ChatOpenAI(model="gpt-4").bind(
            system="""You are an error diagnosis expert.
            Given an error and context, suggest a fix.
            Be specific and actionable."""
        )
    
    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=4, max=10)
    )
    async def execute_with_recovery(self, step_fn, input_data, context):
        try:
            return await step_fn(input_data)
        except Exception as e:
            # Diagnose the error
            diagnosis = await self.error_handler.ainvoke(
                f"Error: {str(e)}\nContext: {context}\nInput: {input_data}"
            )
            
            # Attempt auto-fix based on diagnosis
            if "retry with modified input" in diagnosis.content:
                fixed_input = await self.auto_fix_input(
                    input_data, diagnosis.content
                )
                return await step_fn(fixed_input)
            elif "skip and continue" in diagnosis.content:
                return {"status": "skipped", "reason": diagnosis.content}
            else:
                raise  # Re-raise for retry mechanism
    
    async def auto_fix_input(self, input_data, diagnosis):
        fix_prompt = f"""
        Original input: {input_data}
        Diagnosis: {diagnosis}
        
        Provide corrected input:
        """
        fixed = await self.llm.ainvoke(fix_prompt)
        return fixed.content
```

## Case Study 3: Intelligent Load Balancing

### Challenge

Dynamically route requests to optimal models based on cost, latency, and quality.

<Mermaid chart={`

    Select --> GPT35

import numpy as np
from sklearn.ensemble import RandomForestClassifier

class IntelligentRouter:
    def __init__(self):
        self.models = {
            "gpt-4": {
                "client": ChatOpenAI(model="gpt-4"),
                "cost": 0.03,
                "latency": 2.0,
                "quality": 0.95
            },
            "gpt-3.5-turbo": {
                "client": ChatOpenAI(model="gpt-3.5-turbo"),
                "cost": 0.001,
                "latency": 0.5,
                "quality": 0.80
            },
            "local": {
                "client": ChatOllama(model="llama2"),
                "cost": 0.0,
                "latency": 1.0,
                "quality": 0.70
            }
        }
        
        self.classifier = RandomForestClassifier()
        self.feedback_history = []
    
    def extract_features(self, request: dict) -> np.array:
        return np.array([
            len(request.get("text", "")),  # Length
            request.get("priority", 0.5),   # Priority
            request.get("quality_required", 0.8),  # Quality need
            1 if request.get("sensitive", False) else 0,  # Privacy
        ])
    
    def route(self, request: dict) -> str:
        features = self.extract_features(request)
        
        if len(self.feedback_history) < 100:
            # Rule-based routing for cold start
            if request.get("sensitive"):
                return "local"
            if request.get("quality_required", 0.8) > 0.9:
                return "gpt-4"
            return "gpt-3.5-turbo"
        
        # ML-based routing
        return self.classifier.predict([features])[0]
    
    async def invoke(self, request: dict) -> dict:
        model_name = self.route(request)
        model = self.models[model_name]
        
        start = time.time()
        response = await model["client"].ainvoke(request["text"])
        latency = time.time() - start
        
        # Collect feedback for learning
        self.feedback_history.append({
            "features": self.extract_features(request),
            "model": model_name,
            "latency": latency,
            "cost": model["cost"],
        })
        
        return {
            "response": response.content,
            "model_used": model_name,
            "latency": latency
        }

## Case Study 4: Automated Prompt Optimization

### Challenge

Automatically optimize prompts based on performance metrics.

<Mermaid chart={`
flowchart TB
    Initial[Initial Prompt] --> Test[Test on Dataset]
    Test --> Eval[Evaluate Results]
    Eval --> Score[Calculate Score]
    Score --> Good{Good Enough?}
    Good -->|No| Analyze[Analyze Failures]
    Analyze --> Generate[Generate Variations]
    Generate --> Test
    Good -->|Yes| Deploy[Deploy Prompt]
`} />

```python
class PromptOptimizer:
    def __init__(self, evaluator_llm, target_llm):
        self.evaluator = evaluator_llm
        self.target = target_llm
        self.best_prompt = None
        self.best_score = 0
    
    async def optimize(
        self, 
        initial_prompt: str, 
        test_cases: list[dict],
        iterations: int = 10
    ) -> str:
        current_prompt = initial_prompt
        
        for i in range(iterations):
            # Evaluate current prompt
            score, failures = await self.evaluate(current_prompt, test_cases)
            
            if score > self.best_score:
                self.best_score = score
                self.best_prompt = current_prompt
            
            if score >= 0.95:  # Good enough
                break
            
            # Generate improved prompt
            current_prompt = await self.improve_prompt(
                current_prompt, failures, score
            )
        
        return self.best_prompt
    
    async def evaluate(self, prompt: str, test_cases: list) -> tuple:
        results = []
        failures = []
        
        for case in test_cases:
            response = await self.target.ainvoke(
                prompt.format(**case["input"])
            )
            
            score = await self.score_response(
                response.content, 
                case["expected"]
            )
            results.append(score)
            
            if score < 0.8:
                failures.append({
                    "input": case["input"],
                    "expected": case["expected"],
                    "actual": response.content,
                    "score": score
                })
        
        return np.mean(results), failures
    
    async def improve_prompt(
        self, 
        prompt: str, 
        failures: list, 
        current_score: float
    ) -> str:
        improvement_prompt = f"""
        Current prompt (score: {current_score:.2f}):
        {prompt}
        
        Failed cases:
        {json.dumps(failures[:3], indent=2)}
        
        Analyze why these cases failed and rewrite the prompt
        to handle them better while maintaining performance
        on other cases.
        
        Improved prompt:
        """
        
        result = await self.evaluator.ainvoke(improvement_prompt)
        return result.content
```
