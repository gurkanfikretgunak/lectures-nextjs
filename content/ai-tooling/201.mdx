## LangGraph: Stateful Workflows

LangGraph enables complex, stateful AI workflows with cycles and branching.

<Mermaid chart={`
flowchart LR
    Start[Start] --> Classify[Classify Input]
    Classify --> Simple{Simple?}
    Simple -->|Yes| Direct[Direct Response]
    Simple -->|No| Research[Research]
    Research --> Draft[Draft Response]
    Draft --> Review[Review]
    Review --> Good{Good?}
    Good -->|Yes| End[End]
    Good -->|No| Research
`} />

### Implementation

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict, Literal

class AgentState(TypedDict):
    messages: list
    classification: str
    draft: str
    review_count: int

def classify_input(state: AgentState) -> AgentState:
    # Classify the complexity of the input
    messages = state["messages"]
    # Use LLM to classify
    classification = llm.invoke(f"Classify: {messages[-1]}")
    return {"classification": classification}

def generate_direct(state: AgentState) -> AgentState:
    response = llm.invoke(state["messages"][-1])
    return {"draft": response}

def research(state: AgentState) -> AgentState:
    # Perform research using tools
    results = search_tool.invoke(state["messages"][-1])
    return {"draft": results}

def review(state: AgentState) -> AgentState:
    review_prompt = f"Review this draft: {state['draft']}"
    review_result = llm.invoke(review_prompt)
    return {
        "draft": review_result,
        "review_count": state.get("review_count", 0) + 1
    }

def should_continue(state: AgentState) -> Literal["research", "end"]:
    if "APPROVED" in state["draft"] or state["review_count"] >= 3:
        return "end"
    return "research"

# Build graph
workflow = StateGraph(AgentState)

workflow.add_node("classify", classify_input)
workflow.add_node("direct", generate_direct)
workflow.add_node("research", research)
workflow.add_node("review", review)

workflow.set_entry_point("classify")
workflow.add_conditional_edges(
    "classify",
    lambda s: "direct" if s["classification"] == "simple" else "research"
)
workflow.add_edge("direct", END)
workflow.add_edge("research", "review")
workflow.add_conditional_edges("review", should_continue)

app = workflow.compile()
```

```python

from langchain.globals import set_llm_cache
from langchain_community.cache import RedisSemanticCache

# Simple cache
set_llm_cache(InMemoryCache())

# Semantic cache (finds similar queries)
semantic_cache = RedisSemanticCache(
    redis_url="redis://localhost:6379",
    embedding=OpenAIEmbeddings(),
    score_threshold=0.95  # Similarity threshold
)
```

### Multi-tier Caching

```python
class MultiTierCache:
    def __init__(self):
        self.l1_cache = {}  # In-memory
        self.l2_cache = Redis()  # Redis
        self.semantic = SemanticCache()  # Vector similarity
    
    def get(self, query: str) -> str | None:
        # L1: Exact match in memory
        if query in self.l1_cache:
            return self.l1_cache[query]
        
        # L2: Exact match in Redis
        cached = self.l2_cache.get(query)
        if cached:
            self.l1_cache[query] = cached
            return cached
        
        # L3: Semantic similarity
        similar = self.semantic.find_similar(query, threshold=0.95)
        if similar:
            self.l1_cache[query] = similar
            self.l2_cache.set(query, similar)
            return similar
        
        return None
```

## Security Best Practices

### Input Validation

```python
import re
from pydantic import BaseModel, validator

class QueryInput(BaseModel):
    query: str
    max_tokens: int = 500
    
    @validator("query")
    def validate_query(cls, v):
        # Check length
        if len(v) > 10000:
            raise ValueError("Query too long")
        
        # Block injection attempts
        dangerous_patterns = [
            r"ignore\s+previous\s+instructions",
            r"system\s*:",
            r"<\|.*\|>"
        ]
        for pattern in dangerous_patterns:
            if re.search(pattern, v, re.IGNORECASE):
                raise ValueError("Potentially malicious input")
        
        return v
```

### Output Filtering

```python
class OutputFilter:
    def __init__(self):
        self.pii_patterns = [
            r"\b\d{3}-\d{2}-\d{4}\b",  # SSN
            r"\b\d{16}\b",              # Credit card
        ]
    
    def filter(self, text: str) -> str:
        for pattern in self.pii_patterns:
            text = re.sub(pattern, "[REDACTED]", text)
        return text
```
