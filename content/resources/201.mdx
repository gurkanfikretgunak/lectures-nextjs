### Infrastructure Code

```python
# vLLM Serving Example
from vllm import LLM, SamplingParams

llm = LLM(
    model="meta-llama/Llama-2-7b-chat-hf",
    tensor_parallel_size=2,  # Multi-GPU
    gpu_memory_utilization=0.9
)

sampling_params = SamplingParams(
    temperature=0.7,
    top_p=0.95,
    max_tokens=256
)

outputs = llm.generate(["Hello, how are you?"], sampling_params)
```

### Custom Evaluation

```python
from langsmith import Client
from langsmith.evaluation import evaluate

client = Client()

# Define custom evaluator
def relevance_evaluator(run, example):
    """Check if response is relevant to question."""
    prediction = run.outputs["output"]
    reference = example.outputs["answer"]
    
    # Use LLM to evaluate
    evaluation = llm.invoke(f"""
    Question: {example.inputs['question']}
    Expected topics: {reference}
    Actual response: {prediction}
    
    Is the response relevant? Score 0-1:
    """)
    
    return {"key": "relevance", "score": float(evaluation.content)}

# Run evaluation
results = evaluate(
    lambda inputs: chain.invoke(inputs["question"]),
    data="my-dataset",
    evaluators=[relevance_evaluator]
)
```
