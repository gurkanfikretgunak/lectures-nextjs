from enum import Enum
import re

class IntentType(Enum):
    ORDER_STATUS = "order_status"
    RETURN_REQUEST = "return_request"
    PRODUCT_INFO = "product_info"
    COMPLAINT = "complaint"
    GENERAL_QUERY = "general_query"

@dataclass
class ChatMessage:

    message: str
    timestamp: str
    intent: Optional[IntentType] = None

class CustomerServiceChatbot:
    def __init__(self):

        self.intent_keywords = {
            IntentType.ORDER_STATUS: ["order", "track", "shipping", "delivery", "status"],
            IntentType.RETURN_REQUEST: ["return", "refund", "exchange", "send back"],
            IntentType.PRODUCT_INFO: ["product", "price", "available", "specifications", "features"],
            IntentType.COMPLAINT: ["problem", "issue", "broken", "wrong", "defective"],
            IntentType.GENERAL_QUERY: ["help", "question", "information"]
        }
        self.order_database = {

            "ORD002": {"status": "processing", "tracking": None, "eta": "2024-02-12"}
        }

        if user_id not in self.conversations:

        intent = self._detect_intent(message)
        entities = self._extract_entities(message)

            message=message,
            timestamp="now",
            intent=intent,
            entities=entities
