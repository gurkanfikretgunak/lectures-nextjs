enum HealthStatus {
  EXCELLENT = "excellent",
  GOOD = "good",
  FAIR = "fair",
  POOR = "poor",
  CRITICAL = "critical"
}

enum IssueType {
  DISEASE = "disease",
  PEST = "pest",
  NUTRIENT_DEFICIENCY = "nutrient_deficiency",
  WATER_STRESS = "water_stress",
  WEED_INFESTATION = "weed_infestation"
}

interface SensorReading {
  timestamp: Date;
  temperature: number;
  humidity: number;
  soilMoisture: number;
  phLevel: number;
  nitrogenLevel: number;
  phosphorusLevel: number;
  potassiumLevel: number;
}

interface CropHealth {
  cropId: string;
  fieldId: string;
  healthStatus: HealthStatus;
  issues: IssueType[];
  recommendations: string[];
  riskScore: number;
}

class CropMonitoringSystem {
  private sensorData: Map<string, SensorReading[]> = new Map();
  private thresholds = {
    soilMoistureMin: 30.0,
    soilMoistureMax: 80.0,
    phMin: 6.0,
    phMax: 7.5,
    nitrogenMin: 20.0,
    phosphorusMin: 15.0,
    potassiumMin: 150.0
  };
  
  addSensorReading(fieldId: string, reading: SensorReading): void {
    if (!this.sensorData.has(fieldId)) {
      this.sensorData.set(fieldId, []);
    }
    const readings = this.sensorData.get(fieldId)!;
    readings.push(reading);
    
    // Keep only last 30 days (720 hours)
    if (readings.length > 720) {
      readings.splice(0, readings.length - 720);
    }
  }
  
  analyzeCropHealth(cropId: string, fieldId: string): CropHealth {
    const readings = this.sensorData.get(fieldId);
    
    if (!readings || readings.length === 0) {
      return {
        cropId,
        fieldId,
        healthStatus: HealthStatus.FAIR,
        issues: [],
        recommendations: ["Insufficient data. Please add sensor readings."],
        riskScore: 0.5
      };
    }
    
    const latest = readings[readings.length - 1];
    const issues: IssueType[] = [];
    const recommendations: string[] = [];
    let riskFactors = 0;
    
    // Check soil moisture
    if (latest.soilMoisture < this.thresholds.soilMoistureMin) {
      issues.push(IssueType.WATER_STRESS);
      recommendations.push("Irrigation needed. Soil moisture is below optimal level.");
      riskFactors += 2;
    } else if (latest.soilMoisture > this.thresholds.soilMoistureMax) {
      issues.push(IssueType.WATER_STRESS);
      recommendations.push("Reduce irrigation. Soil moisture is above optimal level.");
      riskFactors += 1;
    }
    
    // Check pH level
    if (latest.phLevel < this.thresholds.phMin) {
      issues.push(IssueType.NUTRIENT_DEFICIENCY);
      recommendations.push("Soil pH is too low. Consider adding lime to raise pH.");
      riskFactors += 1;
    } else if (latest.phLevel > this.thresholds.phMax) {
      issues.push(IssueType.NUTRIENT_DEFICIENCY);
      recommendations.push("Soil pH is too high. Consider adding sulfur to lower pH.");
      riskFactors += 1;
    }
    
    // Check nutrient levels
    if (latest.nitrogenLevel < this.thresholds.nitrogenMin) {
      issues.push(IssueType.NUTRIENT_DEFICIENCY);
      recommendations.push("Nitrogen deficiency detected. Apply nitrogen fertilizer.");
      riskFactors += 2;
    }
    
    if (latest.phosphorusLevel < this.thresholds.phosphorusMin) {
      issues.push(IssueType.NUTRIENT_DEFICIENCY);
      recommendations.push("Phosphorus deficiency detected. Apply phosphorus fertilizer.");
      riskFactors += 1;
    }
    
    if (latest.potassiumLevel < this.thresholds.potassiumMin) {
      issues.push(IssueType.NUTRIENT_DEFICIENCY);
      recommendations.push("Potassium deficiency detected. Apply potassium fertilizer.");
      riskFactors += 1;
    }
    
    // Analyze trends
    if (readings.length >= 7) {
      const recentReadings = readings.slice(-7);
      const moistureTrend = this.calculateTrend(
        recentReadings.map(r => r.soilMoisture)
      );
      
      if (moistureTrend < -5) {
        if (!issues.includes(IssueType.WATER_STRESS)) {
          issues.push(IssueType.WATER_STRESS);
          recommendations.push("Soil moisture is declining rapidly. Increase irrigation frequency.");
          riskFactors += 1;
        }
      }
    }
    
    const riskScore = Math.min(1.0, riskFactors / 10.0);
    
    let healthStatus: HealthStatus;
    if (riskScore >= 0.7) {
      healthStatus = HealthStatus.CRITICAL;
    } else if (riskScore >= 0.5) {
      healthStatus = HealthStatus.POOR;
    } else if (riskScore >= 0.3) {
      healthStatus = HealthStatus.FAIR;
    } else if (riskScore >= 0.1) {
      healthStatus = HealthStatus.GOOD;
    } else {
      healthStatus = HealthStatus.EXCELLENT;
    }
    
    if (issues.length === 0) {
      recommendations.push("Crop health is optimal. Continue current practices.");
    }
    
    return {
      cropId,
      fieldId,
      healthStatus,
      issues,
      recommendations,
      riskScore: Math.round(riskScore * 100) / 100
    };
  }
  
  private calculateTrend(values: number[]): number {
    if (values.length < 2) return 0.0;
    
    const n = values.length;
    const xMean = (n - 1) / 2;
    const yMean = values.reduce((a, b) => a + b, 0) / n;
    
    let numerator = 0;
    let denominator = 0;
    
    for (let i = 0; i < n; i++) {
      numerator += (i - xMean) * (values[i] - yMean);
      denominator += Math.pow(i - xMean, 2);
    }
    
    return denominator === 0 ? 0.0 : numerator / denominator;
  }
  
  predictYield(cropId: string, fieldId: string): {
    predictedYieldKgPerHectare: number;
    confidence: number;
    healthStatus: string;
    riskFactors: number;
  } {
    const health = this.analyzeCropHealth(cropId, fieldId);
    const baseYield = 1000; // kg per hectare
    
    const yieldMultipliers: Record<HealthStatus, number> = {
      [HealthStatus.EXCELLENT]: 1.2,
      [HealthStatus.GOOD]: 1.1,
      [HealthStatus.FAIR]: 1.0,
      [HealthStatus.POOR]: 0.8,
      [HealthStatus.CRITICAL]: 0.6
    };
    
    const predictedYield = baseYield * yieldMultipliers[health.healthStatus];
    
    return {
      predictedYieldKgPerHectare: Math.round(predictedYield * 100) / 100,
      confidence: Math.round((1.0 - health.riskScore) * 100) / 100,
      healthStatus: health.healthStatus,
      riskFactors: health.issues.length
    };
  }
}

from typing import Dict, List
from dataclasses import dataclass
from enum import Enum

class FieldZone(Enum):
    HIGH_PRODUCTIVITY = "high"
    MEDIUM_PRODUCTIVITY = "medium"
    LOW_PRODUCTIVITY = "low"

@dataclass
class FieldZoneData:
    zone_id: str
    zone_type: FieldZone
    area_hectares: float
    soil_quality: float
    historical_yield: float
    recommended_fertilizer: float
    recommended_seeding_rate: float

class PrecisionAgricultureSystem:
    def __init__(self):
        self.field_zones: Dict[str, FieldZoneData] = {}
        self.base_fertilizer_rate = 150  # kg/hectare
        self.base_seeding_rate = 200  # seeds/hectare
    
    def create_zone(self, zone_id: str, soil_quality: float, 
                   historical_yield: float, area: float):
        """Create a field zone with analysis"""
        # Classify zone based on productivity
        if historical_yield >= 8.0:  # tons/hectare
            zone_type = FieldZone.HIGH_PRODUCTIVITY
        elif historical_yield >= 6.0:
            zone_type = FieldZone.MEDIUM_PRODUCTIVITY
        else:
            zone_type = FieldZone.LOW_PRODUCTIVITY
        
        # Calculate variable rate recommendations
        yield_factor = historical_yield / 7.0  # Normalize to average
        soil_factor = soil_quality / 0.7  # Normalize to good soil
        
        # Adjust fertilizer based on zone productivity
        fertilizer_rate = self.base_fertilizer_rate * yield_factor * soil_factor
        
        # Adjust seeding rate
        seeding_rate = self.base_seeding_rate * yield_factor
        
        zone = FieldZoneData(
            zone_id=zone_id,
            zone_type=zone_type,
            area_hectares=area,
            soil_quality=soil_quality,
            historical_yield=historical_yield,
            recommended_fertilizer=round(fertilizer_rate, 2),
            recommended_seeding_rate=round(seeding_rate, 0)
        )
        
        self.field_zones[zone_id] = zone
        return zone
    
    def optimize_field_plan(self, field_id: str) -> Dict:
        """Generate optimized field plan"""
        zones = [z for z in self.field_zones.values() 
                if z.zone_id.startswith(field_id)]
        
        if not zones:
            return {"error": "No zones found for field"}
        
        total_area = sum(z.area_hectares for z in zones)
        total_fertilizer = sum(
            z.recommended_fertilizer * z.area_hectares for z in zones
        )
        total_seeds = sum(
            z.recommended_seeding_rate * z.area_hectares for z in zones
        )
        
        return {
            "field_id": field_id,
            "total_area_hectares": round(total_area, 2),
            "total_fertilizer_kg": round(total_fertilizer, 2),
            "total_seeds": round(total_seeds, 0),
            "zones": [
                {
                    "zone_id": z.zone_id,
                    "type": z.zone_type.value,
                    "area": z.area_hectares,
                    "fertilizer_rate": z.recommended_fertilizer,
                    "seeding_rate": z.recommended_seeding_rate
                }
                for z in zones
            ],
            "savings_estimate": self._calculate_savings(zones)
        }
    
    def _calculate_savings(self, zones: List[FieldZoneData]) -> Dict:
        """Calculate estimated savings from precision agriculture"""
        uniform_fertilizer = sum(
            self.base_fertilizer_rate * z.area_hectares for z in zones
        )
        variable_fertilizer = sum(
            z.recommended_fertilizer * z.area_hectares for z in zones
        )
        
        fertilizer_savings = uniform_fertilizer - variable_fertilizer

            "savings_percent": round(savings_percent, 1),
            "estimated_cost_savings_usd": round(fertilizer_savings * 0.5, 2)  # $0.50/kg

interface FieldZoneData {
  zoneId: string;
  zoneType: FieldZone;
  areaHectares: number;
  soilQuality: number;
  historicalYield: number;
  recommendedFertilizer: number;
  recommendedSeedingRate: number;
}

class PrecisionAgricultureSystem {
  private fieldZones: Map<string, FieldZoneData> = new Map();
  private baseFertilizerRate = 150; // kg/hectare
  private baseSeedingRate = 200; // seeds/hectare
  
  createZone(
    zoneId: string,
    soilQuality: number,
    historicalYield: number,
    area: number
  ): FieldZoneData {
    let zoneType: FieldZone;
    if (historicalYield >= 8.0) {
      zoneType = FieldZone.HIGH_PRODUCTIVITY;
    } else if (historicalYield >= 6.0) {
      zoneType = FieldZone.MEDIUM_PRODUCTIVITY;
    } else {
      zoneType = FieldZone.LOW_PRODUCTIVITY;
    }
    
    const yieldFactor = historicalYield / 7.0;
    const soilFactor = soilQuality / 0.7;
    
    const fertilizerRate = this.baseFertilizerRate * yieldFactor * soilFactor;
    const seedingRate = this.baseSeedingRate * yieldFactor;
    
    const zone: FieldZoneData = {
      zoneId,
      zoneType,
      areaHectares: area,
      soilQuality,
      historicalYield,
      recommendedFertilizer: Math.round(fertilizerRate * 100) / 100,
      recommendedSeedingRate: Math.round(seedingRate)
    };
    
    this.fieldZones.set(zoneId, zone);
    return zone;
  }
  
  optimizeFieldPlan(fieldId: string): {
    fieldId: string;
    totalAreaHectares: number;
    totalFertilizerKg: number;
    totalSeeds: number;
    zones: Array<{
      zoneId: string;
      type: string;
      area: number;
      fertilizerRate: number;
      seedingRate: number;
    }>;
    savingsEstimate: {
      fertilizerSavingsKg: number;
      savingsPercent: number;
      estimatedCostSavingsUsd: number;
    };
  } {
    const zones = Array.from(this.fieldZones.values())
      .filter(z => z.zoneId.startsWith(fieldId));
    
    const totalArea = zones.reduce((sum, z) => sum + z.areaHectares, 0);
    const totalFertilizer = zones.reduce(
      (sum, z) => sum + z.recommendedFertilizer * z.areaHectares, 0
    );
    const totalSeeds = zones.reduce(
      (sum, z) => sum + z.recommendedSeedingRate * z.areaHectares, 0
    );
    
    const uniformFertilizer = zones.reduce(
      (sum, z) => sum + this.baseFertilizerRate * z.areaHectares, 0
    );
    const fertilizerSavings = uniformFertilizer - totalFertilizer;
    const savingsPercent = uniformFertilizer > 0
      ? (fertilizerSavings / uniformFertilizer) * 100
      : 0;
    
    return {
      fieldId,
      totalAreaHectares: Math.round(totalArea * 100) / 100,
      totalFertilizerKg: Math.round(totalFertilizer * 100) / 100,
      totalSeeds: Math.round(totalSeeds),
      zones: zones.map(z => ({
        zoneId: z.zoneId,
        type: z.zoneType,
        area: z.areaHectares,
        fertilizerRate: z.recommendedFertilizer,
        seedingRate: z.recommendedSeedingRate
      })),
      savingsEstimate: {
        fertilizerSavingsKg: Math.round(fertilizerSavings * 100) / 100,
        savingsPercent: Math.round(savingsPercent * 10) / 10,
        estimatedCostSavingsUsd: Math.round(fertilizerSavings * 0.5 * 100) / 100
      }
    };
  }
}

// Usage
const system = new PrecisionAgricultureSystem();
system.createZone("FIELD1_ZONE1", 0.8, 9.0, 5.0);
system.createZone("FIELD1_ZONE2", 0.6, 6.5, 8.0);
system.createZone("FIELD1_ZONE3", 0.5, 5.0, 3.0);

const fieldPlan = system.optimizeFieldPlan("FIELD1");
console.log(\`Field Plan: \${fieldPlan.fieldId}\`);
console.log(\`Total Area: \${fieldPlan.totalAreaHectares} hectares\`);
console.log(\`Estimated Savings: $\${fieldPlan.savingsEstimate.estimatedCostSavingsUsd}\`);`}
